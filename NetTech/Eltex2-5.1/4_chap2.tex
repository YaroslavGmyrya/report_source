\section{Настройка интерфейсов роутеров}

\begin{lstlisting}[language=C, caption={Пример настройки интерфейсов роутера}]
# R1
configure

interface gigabitethernet 1/0/1
ip address 192.168.100.1/30
ip firewall disable
exit

interface gigabitethernet 1/0/2
ip address 10.10.0.1/24
ip firewall disable
exit

interface gigabitethernet 1/0/3
ip address 10.10.1.1/24
ip firewall disable
exit

interface gigabitethernet 1/0/4
ip address 10.10.100.1/30
ip firewall disable
exit

do commit
do confirm

# R2
configure

interface gigabitethernet 1/0/1
ip address 192.168.100.2/30
ip firewall disable
exit

interface gigabitethernet 1/0/2
ip address 10.10.10.1/24
ip firewall disable
exit

interface gigabitethernet 1/0/3
ip address 10.10.11.1/24
ip firewall disable
exit

interface gigabitethernet 1/0/4
ip address 172.16.200.2/30
ip firewall disable
exit

do commit
do confirm

# R3
configure

interface gigabitethernet 1/0/1
ip address 172.16.200.1/30
ip firewall disable
exit

interface gigabitethernet 1/0/2
ip address 10.10.100.2/30
ip firewall disable
exit 


do commit
do confirm
\end{lstlisting}
