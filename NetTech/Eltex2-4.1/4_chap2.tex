\chapter{Практика}

\section{Создание и настройка vlan}

\begin{lstlisting}[language=C, caption={Пример создания/настройки vlan}]
configure

# создаем vlan
vlan 10
# задаем имя
name "nw 10.10.10.0"

exit

vlan 20
name "nw 10.20.20.0"

do commit
do confirm
\end{lstlisting}

\section{Добавление интерфейсов в vlan}

\begin{lstlisting}[language=C, caption={Пример создания/настройки vlan}]
configure

# выбираем интерфейсы
interface gigabitethernet 1/0/1-2

# переводим интерфейс в режим порта
mode switchport

# переводим порт в режим доступа
switchport mode access

# добавляем в vlan
switchport access vlan 10

# то же самое для других интерфейсов
interface gigabitethernet 1/0/11-12
mode switchport
switchport mode access
switchport access vlan 20

# принимаем изменения
do commit
do confirm

\end{lstlisting}
