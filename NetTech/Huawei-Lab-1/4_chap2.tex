\section{Настройка интерфейсов роутера}

\begin{lstlisting}[language=C, caption={Пример настройки интерфейсов роутера}]

# переход в режим конфигурации
<Huawei>system-view
# выбор интерфейса
[Datacom-Router]interface GigabitEthernet 0/0/1
# назначение IP и маски
[Datacom-Router]ip address 192.168.1.1 24
\end{lstlisting}
