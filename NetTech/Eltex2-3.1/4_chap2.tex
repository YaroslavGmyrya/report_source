\section{Настройка имени}

\begin{lstlisting}[language=C, caption={Пример настройки имени устройства}]
# переход в режим конфигурации
configure
# задаем имя
hostname R1
# принимаем настройки
do commit
do confirm
\end{lstlisting}

\section{Настройка времени}

\begin{lstlisting}[language=C, caption={Пример настройки времени}]
# В режиме пользователя
set date 16:49:00 25 March 2026
# вносим изменения
commit
# проверка
show date
# переходим в режим конфигурации
configure
# задаем часовой пояс
clock timezone gmt +7
# вносим изменения
do commit
do confirm
\end{lstlisting}

\section{Настройка IP на интерфейсе}

\begin{lstlisting}[language=C, caption={Пример настройки IP на интерфейсе}]
# переход в режим конфигурации
configure
# переход к настройке интерфейса gi 1/0/1
interface gigabitethernet 1/0/1
# задаем ip с маской
ip address 192. 168. 100. 1/24
# отключаем firewall
ip firewall disable
# подтверждаем изменения
do commit
do confirm
\end{lstlisting}

\section{Создание и настройка Loopback интерфейса}

\begin{lstlisting}[language=C, caption={Пример создания и настройки Loopback интерфейса}]
# переход в режим конфигурации
configure
# переход к настройке интерфейса Loopback 1
interface Loopback 1
# задаем ip с маской
ip address 8.8.8.8/32
exit
do commit
do confirm
\end{lstlisting}

\section{Создание и настройка sub интерфейса}

\begin{lstlisting}[language=C, caption={Пример создания и настройки sub интерфейса}]
configure
# переходим в настройку интерфейса gi 1/0/1. Через точку указан vlan для sub интерфейса
interface gigabitethernet 1/0/1.50
# задаем ip
ip address 172. 16. 25. 1/24
# отключаем firewall
ip firewall disable
\end{lstlisting}
