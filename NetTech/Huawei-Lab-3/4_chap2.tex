\chapter{Практика}


\section{Настройка интерфейсов роутера}

\begin{lstlisting}[language=C, caption={Пример настройки интерфейсов роутера}]
# R1
[R1]interface GigabitEthernet0/0/1
[R1-GigabitEthernet0/0/1]ip address 10.1.2.1 24

#R2
[R2]interface GigabitEthernet0/0/2
[R2-GigabitEthernet0/0/2]ip address 10.1.10.1 24

# сохраняем настройки
save
\end{lstlisting}

\section{Настройка портов коммутаторов}

\begin{lstlisting}[language=C, caption={Пример настройки портов коммутатора}]
# создаем vlan 3
[S3]vlan 3
# переводим порт в режим досутпа и устанавливаем vlan3
[S3]interface GigabitEthernet0/0/1
[S3-GigabitEthernet0/0/1]port link-type access
[S3-GigabitEthernet0/0/1]port default vlan 3
[S3-GigabitEthernet0/0/1]quit
# сохраняем настройки
save
\end{lstlisting}

Аналогичная настройка для S4
\section{Создание и настройка SVI интерфейса}

\begin{lstlisting}[language=C, caption={Пример создания и настройки SVI интерфейса}]
[S3] interface Vlanif 3
[S3-Vlanif3]ip address 10.1.3.2 24
\end{lstlisting}

Аналогично для S4
\section{Настройка trunk порта между коммутаторами}

\begin{lstlisting}[language=C, caption={Пример настройки trunk порта}]
# выбираем интерфейс
[S2]interface GigabitEthernet0/0/10
# создаем транк порт
[S2-GigabitEthernet0/0/10]port link-type trunk
# разрешаем vlan 2 3
[S2-GigabitEthernet0/0/10]port trunk allow-pass vlan 2 3

\end{lstlisting}
